One-Time Supabase SQL Setup (Normalized Per-Record Tables + RLS + Storage) 
create table if not exists public.pm_reports (
  id text primary key,
  customer_account text,
  comm_no text,
  visit_date text,
  quarter text,
  claim_period text,
  technician_name text,
  updated_at timestamptz default now(),
  payload jsonb not null
);

create table if not exists public.service_logs (
  id text primary key,
  pm_report_id text,
  customer_account text,
  comm_no text,
  visit_date text,
  visit_type text,
  quarter text,
  claim_period text,
  engineer_name text,
  remarks text,
  file_name text,
  file_url text,
  updated_at timestamptz default now(),
  payload jsonb not null
);

create table if not exists public.invoice_requests (
  id text primary key,
  customer_account text,
  comm_no text,
  request_date text,
  invoice_type text,
  invoice_value numeric default 0,
  quarter text,
  claim_period text,
  invoice_from text,
  po_number text,
  requested_by text,
  updated_at timestamptz default now(),
  payload jsonb not null
);

create table if not exists public.folder_files (
  id text primary key,
  customer_account text,
  comm_no text,
  subfolder text,
  category text,
  file_name text,
  file_url text,
  updated_at timestamptz default now(),
  payload jsonb not null
);

create table if not exists public.machine_records (
  comm_no text primary key,
  customer_account text,
  record_kind text default 'override',
  updated_at timestamptz default now(),
  payload jsonb not null
);

create table if not exists public.hanel_amc_state (
  id text primary key,
  updated_at timestamptz default now(),
  payload jsonb not null
);

do $$
declare
  t text;
begin
  foreach t in array array['pm_reports','service_logs','invoice_requests','folder_files','machine_records','hanel_amc_state']
  loop
    execute format('alter table public.%I enable row level security;', t);
    execute format('drop policy if exists "Allow access %I" on public.%I;', t, t);
    execute format('create policy "Allow access %I" on public.%I for all using (true) with check (true);', t, t);
  end loop;
end $$;

insert into storage.buckets (id, name, public)
